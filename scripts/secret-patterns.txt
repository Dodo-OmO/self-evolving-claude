# secret-patterns.txt
# 推送到公开仓前要扫的敏感模式 —— 每行一个正则（PowerShell -match 语法），'#' 注释。
# 按你的真实情况填：真名 / 化名、邮箱、内网或本机绝对路径、私有项目代号、token 格式等。
# 下面是占位示例，请替换成你自己的，再启用 check-secrets.ps1。

# 邮箱
[a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+\.(com|net|org|cn|io)

# token / key 常见前缀
(sk-|ghp_|gho_|xox[baprs]-)[A-Za-z0-9]{10,}

# —— 下面按需取消注释并替换成你的 ——
# 你的真名 / 化名
# Zhang San|zhangsan
# 本机绝对路径前缀
# [A-Z]:\\Users\\
# 私有项目内部代号
# my-private-project-codename
